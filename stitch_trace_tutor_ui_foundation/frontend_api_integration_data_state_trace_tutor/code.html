api Axios Client & JWT Interceptors

client.ts

The centralized HTTP client handles automatic token injection, queueing requests during token refresh cycles, and normalizing Django Rest Framework (DRF) error payloads. 

import axios from 'axios';

export const apiClient = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL || 'https://api.tracetutor.edu/v1/',
  timeout: 15000,
  headers: { 'Content-Type': 'application/json' }
});

apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('access_token');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});
