Expected Reference Implementation
def find_max(arr: list[int]) -> int:
    if not arr: raise ValueError("Empty array")
    max_val = arr[0]
    for i in range(1, len(arr)):
        if arr[i] > max_val:
            max_val = arr[i]
    return max_val
